cmake_minimum_required(VERSION 3.20)
project(goto-vv LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SLANG_SOURCE "" CACHE PATH "Checkout of slang v11.0")
if(SLANG_SOURCE)
  set(SLANG_INCLUDE_TESTS OFF CACHE BOOL "")
  set(SLANG_INCLUDE_INSTALL OFF CACHE BOOL "")
  set(SLANG_USE_MIMALLOC OFF CACHE BOOL "")
  add_subdirectory("${SLANG_SOURCE}" slang EXCLUDE_FROM_ALL)
else()
  find_package(slang 11.0.0 EXACT CONFIG REQUIRED)
endif()
get_filename_component(EBMC_SRC "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(CBMC "${EBMC_SRC}/../lib/cbmc/src")
add_executable(goto-vv main.cpp compiler.cpp expressions.cpp statements.cpp constrain.cpp clock_gate_contract.cpp)
target_include_directories(goto-vv PRIVATE "${EBMC_SRC}" "${CBMC}")
target_compile_definitions(goto-vv PRIVATE "LOCAL_IREP_IDS=<hw_cbmc_irep_ids.h>")
target_link_libraries(goto-vv PRIVATE slang::slang
  "${EBMC_SRC}/verilog/verilog.a" "${CBMC}/util/util.a" "${CBMC}/big-int/big-int.a")
